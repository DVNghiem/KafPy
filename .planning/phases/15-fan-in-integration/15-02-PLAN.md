pub fn register_into_consumer(self, py_consumer: &mut PyConsumer) -> FanInRegistration {

        Arc::clone(&self.callback),
        None, // retry_policy
        self.mode.clone(),
        None, // batch_policy
        timeout,
        self.handler_key.clone(),
        None, // rayon_pool

        Arc::new(handler),

}
